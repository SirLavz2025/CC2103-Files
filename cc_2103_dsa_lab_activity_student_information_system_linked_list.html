Complete Starter Implementation

studentinfo_singlylinkedlist.java JDK 11+ Ready 
import java.util.Scanner;

 package studentinfo_singlylinkedlist;

import java.util.Scanner;

public class StudentInfo_SinglyLinkedList {

    // Step 1: Self-Referential Node Structure (Made static so main can access it)
    static class StudentNode {
        int id;
        String name;
        double gpa;
        StudentNode next; // Pointer to the next student node in memory

        public StudentNode(int id, String name, double gpa) {
            this.id = id;
            this.name = name;
            this.gpa = gpa;
            this.next = null;
        }
    }

    // Step 2: Custom Singly Linked List Implementation (Made static)
    static class StudentLinkedList {
        private StudentNode head; // Points to the first node in the list

        public StudentLinkedList() {
            this.head = null;
        }

        // Append student to the tail of the list
        public void addStudent(int id, String name, double gpa) {
            StudentNode newNode = new StudentNode(id, name, gpa);

            if (head == null) {
                head = newNode;
                System.out.println("--> Success: Added as Head Node.");
                return;
            }

            StudentNode current = head;
            while (current.next != null) {
                current = current.next;
            }
            current.next = newNode;
            System.out.println("--> Success: Appended to List.");
        }

        // Traverse and display all student records
        public void displayStudents() {
            if (head == null) {
                System.out.println("--> List is empty!");
                return;
            }

            System.out.println("\n--- Current Student Records ---");
            StudentNode current = head;
            while (current != null) {
                System.out.println("ID: " + current.id + " | Name: " + current.name + " | GPA: " + current.gpa);
                current = current.next;
            }
        }

        // Search for a student by ID
        public void searchStudent(int id) {
            StudentNode current = head;
            while (current != null) {
                if (current.id == id) {
                    System.out.println("--> Record Found: " + current.name + " (GPA: " + current.gpa + ")");
                    return;
                }
                current = current.next;
            }
            System.out.println("--> Record with ID " + id + " not found.");
        }

        // Delete a student node by ID
        public void deleteStudent(int id) {
            if (head == null) {
                System.out.println("--> Error: List is empty.");
                return;
            }

            // Case 1: Target node is Head
            if (head.id == id) {
                head = head.next;
                System.out.println("--> Student " + id + " deleted (Head updated).");
                return;
            }

            // Case 2: Target node is inside or at tail
            StudentNode current = head;
            StudentNode prev = null;

            while (current != null && current.id != id) {
                prev = current;
                current = current.next;
            }

            if (current == null) {
                System.out.println("--> Record not found.");
            } else {
                prev.next = current.next;
                System.out.println("--> Student " + id + " deleted.");   
            }
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        // Corrected: Use custom StudentLinkedList instead of java.util.LinkedList
        StudentLinkedList list = new StudentLinkedList(); 
        boolean running = true;

        System.out.println("=========================================");
        System.out.println("   STUDENT LINKED LIST MANAGEMENT SYSTEM ");
        System.out.println("=========================================");

        while (running) {
            System.out.println("\n--- MENU ---");
            System.out.println("1. Add Student");
            System.out.println("2. Display All Students");
            System.out.println("3. Search Student by ID");
            System.out.println("4. Delete Student by ID");
            System.out.println("5. Exit");
            System.out.print("Select an option (1-5): ");

            if (!scanner.hasNextInt()) {
                System.out.println("--> Invalid input. Please enter a number.");
                scanner.next(); // Clear invalid token
                continue;
            }

            int choice = scanner.nextInt();

            switch (choice) {
                case 1:
                    System.out.print("Enter Student ID: ");
                    while (!scanner.hasNextInt()) {
                        System.out.println("--> Invalid ID. Must be an integer.");
                        scanner.next();
                    }
                    int id = scanner.nextInt();
                    scanner.nextLine(); // Consume newline

                    System.out.print("Enter Student Name: ");
                    String name = scanner.nextLine();

                    System.out.print("Enter GPA: ");
                    while (!scanner.hasNextDouble()) {
                        System.out.println("--> Invalid GPA. Must be a decimal number.");
                        scanner.next();
                    }
                    double gpa = scanner.nextDouble();

                    list.addStudent(id, name, gpa);
                    break;

                case 2:
                    list.displayStudents();
                    break;

                case 3:
                    System.out.print("Enter Student ID to Search: ");
                    while (!scanner.hasNextInt()) {
                        System.out.println("--> Invalid ID.");
                        scanner.next();
                    }
                    int searchId = scanner.nextInt();
                    list.searchStudent(searchId);
                    break;

                case 4:
                    System.out.print("Enter Student ID to Delete: ");
                    while (!scanner.hasNextInt()) {
                        System.out.println("--> Invalid ID.");
                        scanner.next();
                    }
                    int deleteId = scanner.nextInt();
                    list.deleteStudent(deleteId);
                    break;

                case 5:
                    running = false;
                    System.out.println("\nExiting application. Goodbye!");
                    break;

                default:
                    System.out.println("--> Invalid option. Please choose between 1 and 5.");
            }
        }

        scanner.close();
    }
}

Key Components Breakdown

StudentNode Class

Holds student data fields along with `next` which acts as a pointer storing the address of the subsequent node.

The Head Pointer

Points strictly to the first element. If `head == null`, the memory list contains zero elements. Losing `head` causes memory reference loss.

Deletion Logic Edge Cases

Requires updating pointers. Deleting `head` moves head pointer down (`head = head.next`). Deleting interior nodes links `prev.next` directly to `current.next`.
