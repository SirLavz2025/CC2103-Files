import java.util.Random;

public class PerformanceAnalyzer {

    // Generates an array of random integers of size n
    public static int[] generateData(int size) {
        Random rand = new Random();
        int[] data = new int[size];
        for (int i = 0; i < size; i++) {
            data[i] = rand.nextInt(size);
        }
        return data;
    }

    // Algorithm A: Linear Search
    public static boolean linearSearch(int[] arr, int target) {
        for (int i = 0; i < arr.length; i++) {
            if (arr[i] == target) {
                return true;
            }
        }
        return false;
    }

    // Algorithm B: Cumulative Sum
    public static long calculateSum(int[] arr) {
        long sum = 0;
        for (int i = 0; i < arr.length; i++) {
            sum += arr[i];
        }
        return sum;
    }

    public static void main(String[] args) {
        int[] testSizes = {10000, 100000, 1000000};

        for (int size : testSizes) {
            int[] dataset = generateData(size);
            int target = -1; // Worst-case scenario for search

            long startTime = System.nanoTime();
            linearSearch(dataset, target);
            long endTime = System.nanoTime();
            long searchTime = endTime - startTime;

            startTime = System.nanoTime();
            calculateSum(dataset);
            endTime = System.nanoTime();
            long sumTime = endTime - startTime;

            System.out.println("Dataset Size: " + size);
            System.out.println("  - Linear Search Time: " + searchTime + " ns");
            System.out.println("  - Sum Calculation Time: " + sumTime + " ns");
            System.out.println("----------------------------------------");
        }
    }
}
